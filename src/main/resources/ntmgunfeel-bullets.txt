# NTM Reimagined Arms: the mod's own bullet rules (syntax: config/ntm-gunfeel-bullets.txt, whose lines win)
break minecraft:pointed_dripstone,minecraft:flower_pot,minecraft:decorated_pot instant drop
break #minecraft:leaves never through
# plants a player bumps into (the rest have no collision and are flown through anyway): bullets fly through, no mark
break minecraft:azalea,minecraft:flowering_azalea,minecraft:bamboo,minecraft:big_dripleaf,minecraft:lily_pad,minecraft:pitcher_crop,minecraft:moss_carpet,minecraft:pale_moss_carpet,minecraft:cactus,minecraft:cocoa,minecraft:sea_pickle,minecraft:chorus_plant,minecraft:chorus_flower,minecraft:mangrove_roots never through
# wood: planks and what is made of them (not chests, barrels, shelves and other storage: they would spill)
# break once the bullets' damage on it adds up to 45 x its hardness (planks 90, doors 135), no drops, then the bullet flies on;
# a bullet pierces it keeping 0.8 of its damage less 1.5 x hardness (planks: 20 -> 13 -> 7.4 -> 2.9, so 3 planks)
break #minecraft:planks,#minecraft:wooden_slabs,#minecraft:wooden_stairs,#minecraft:wooden_fences,#minecraft:fence_gates,#minecraft:wooden_doors,#minecraft:wooden_trapdoors,#minecraft:wooden_buttons,#minecraft:wooden_pressure_plates,#minecraft:all_signs,minecraft:bamboo_mosaic,minecraft:crafting_table,minecraft:bookshelf,minecraft:ladder,minecraft:composter,minecraft:note_block,minecraft:loom,minecraft:cartography_table,minecraft:fletching_table,minecraft:smithing_table damage 45 pass pierce=1.5 keep=0.8
# glass and panes (vanilla, any colour, tinted, and other mods' glass tagged so): break after 10 damage whatever the
# bullet, then it flies on; a bullet goes through as through planks, keeping 0.8 of its damage less 1.5 x hardness (0.3)
break #c:glass_blocks,#c:glass_panes hp 10 pass pierce=1.5 keep=0.8
# NTM's glass: breaks after the bullets' damage on it adds up (cracks show), then the bullet flies on; till then a
# bullet goes through by the glass's material (material lines below). Quartz glass as a window; the special glass
# (boron, lead, uranium...) holds more; wired glass far more; the bulletproof laminate takes a long burst
break hbm:glass_quartz hp 10 pass
break hbm:glass_boron,hbm:glass_lead,hbm:glass_polarized,hbm:glass_polonium,hbm:glass_trinitite,hbm:glass_uranium,hbm:glass_ash hp 25 pass
break hbm:reinforced_glass,hbm:reinforced_glass_pane hp 120 pass
break hbm:reinforced_laminate,hbm:reinforced_laminate_pane hp 400 pass
break minecraft:melon,minecraft:pumpkin,minecraft:carved_pumpkin,minecraft:jack_o_lantern hits 4 drop
hit * particles=block*6,minecraft:smoke*1 sound=block/0.6 range=40
# by the sound type first: what the material lines below leave, blocks a bullet never pierces (bedrock, NTM's
# hardened bricks and crystals, statues), keeps it
hit @stone,@deepslate,@deepslate_bricks,@deepslate_tiles,@polished_deepslate,@tuff,@tuff_bricks,@polished_tuff,@calcite,@basalt,@netherrack,@nether_bricks,@nether_ore,@nether_gold_ore,@dripstone_block,@pointed_dripstone,@mud_bricks,@bone_block,@ancient_debris,@gilded_blackstone,@amethyst particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.stone/0.8 range=40
hit @metal,@copper,@copper_bulb,@copper_grate,@chain,@anvil,@heavy_core,@iron,@lantern,@netherite_block,@vault,@trial_spawner particles=block*4,minecraft:crit*2 sound=ntmgunfeel:bullet.hit.metal/0.8 range=40
# hit sounds by the block's material ($<material>: the same one its piercing and marks use; bullet/hit/, see
# sounds/bullet/CREDITS.txt); later lines win
hit $wood particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.wood/0.8 range=40
hit $soft,$soil particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.dirt/0.8 range=40
# gravel crunches
hit minecraft:gravel,minecraft:suspicious_gravel particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.gravel/0.8 range=40
hit $stone particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.stone/0.8 range=40
hit $concrete particles=block*6,minecraft:smoke*1 sound=ntmgunfeel:bullet.hit.concrete/0.8 range=40
hit $metal particles=block*4,minecraft:crit*2 sound=ntmgunfeel:bullet.hit.metal/0.8 range=40
hit $steel,$machine particles=block*4,minecraft:crit*2 sound=ntmgunfeel:bullet.hit.steel/0.8 range=40
# glass: its own glass sound, more shards
hit $glass,$tough_glass,$reinforced_glass,$laminate particles=block*8 sound=block/0.8 range=40
# what a bullet meets in a block, for penetration: material <blocks> none|glass|tough_glass|reinforced_glass|laminate|wood|soft|stone|concrete|metal|steel|machine|unbreakable (metal = soft metals)
# without a line the mod works it out from the block (tags, class, sound, name, blast resistance: Materials); later lines win;
# the jar's ntmgunfeel-materials.txt (the block review) comes before these. A bullet pierces a block by its material
# (as NTM's armour) unless a break line gives it pierce=
# brick-built machines: brick through and through, not a hollow steel casing
material hbm:foundry_basin,hbm:foundry_channel,hbm:foundry_mold,hbm:foundry_outlet,hbm:foundry_slagtap,hbm:foundry_tank,hbm:furnace_combination,hbm:machine_crucible,hbm:machine_blast_furnace,hbm:machine_furnace_brick,hbm:chimney_brick,hbm:chimney_industrial,hbm:machine_electrolyser,hbm:machine_tower_small,hbm:machine_tower_large concrete
# glass: the special glass holds the weak pistols, wired glass more, the bulletproof laminate all but the strong rounds
material hbm:glass_boron,hbm:glass_lead,hbm:glass_polarized,hbm:glass_polonium,hbm:glass_trinitite,hbm:glass_uranium,hbm:glass_ash tough_glass
material hbm:reinforced_glass,hbm:reinforced_glass_pane reinforced_glass
material hbm:reinforced_laminate,hbm:reinforced_laminate_pane laminate
# they sound like glass but are not: plastic floor tiles, ceramic lab tiles
material hbm:vinyl_tile_large,hbm:vinyl_tile_small wood
material hbm:tile_lab,hbm:tile_lab_cracked,hbm:tile_lab_broken stone
# bullet marks (textures/effect, made on the decal generator page: review/decals) come from the block's material,
# the same one its piercing uses: stone and concrete chip (spall, gouge, crater), glass cracks (shatter, lines, star),
# soft ground takes a soft hole (soot, burn, blot), metal, steel and machines keep the old holes (old, lines, star),
# the rest the old holes, a gouge, a blot. A decal line changes that for its blocks, each style from the last line
# that names it: hole = old | spall | shatter | soot; scratch (a ricochet) = gouge | burn | lines;
# scorch (explosive and incendiary rounds) = star | blot | crater
# a grazing bullet bounces off hard blocks (NTM's ricochet) and digs into soft ground and wood; glass it breaks
# or goes through. By the block's material, as the hit sounds
ricochet $soft,$soil,$wood,$glass,$tough_glass,$reinforced_glass,$laminate,$none no
# fences, fence gates and walls: a bullet meets their posts and rails, not their 1.5 block high hitbox (by their class,
# any mod's). NTM's chain-link fence is a fence by its class, but goes by the pixels of its mesh
path hbm:fence_metal,hbm:fence_metal_post pixels
# holes on glass (any mod's block with glass in its id) follow only the hitbox: its see-through pixels are still glass
hole ~.*glass.* hitbox
whizz sound=ntmgunfeel:bullet.whizz/0.9 radius=3 crack=ntmgunfeel:bullet.crack/0.7 crackradius=1.2
